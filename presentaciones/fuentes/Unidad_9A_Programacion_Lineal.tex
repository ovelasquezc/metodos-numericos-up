\begin{frame}{El plan de producción de BIM}
\begin{columns}[c,onlytextwidth]
\begin{column}{.46\textwidth}
BIM fabrica dos tipos de chips. Cada chip de tipo 1 deja un margen de $12$
y cada chip de tipo 2 deja un margen de $9$.
\medskip

Los insumos son limitados. Un chip de tipo 1 usa el doble de cobre que
uno de tipo 2.
\medskip

\concepto{\textquestiondown{}Qué mezcla maximiza el margen total?}
\end{column}
\begin{column}{.51\textwidth}

\begin{tabular}{lrrr}

Insumo & Tipo 1 & Tipo 2 & Límite\\
\midrule
Silicio   & 1 & 0 & $1000$\\
Germanio  & 0 & 1 & $1500$\\
Plástico  & 1 & 1 & $1750$\\
Cobre     & 4 & 2 & $4800$\\
\bottomrule
\end{tabular}
\medskip

\raggedright Los coeficientes indican el consumo por chip en las unidades
de cada insumo.

\begin{frame}{Variables, objetivo y restricciones}
$x_1,x_2$ son las cantidades de chips de cada tipo. El modelo continuo es
\[
\begin{aligned}
\max \quad &12x_1+9x_2\\
\text{sujeto a}\quad
 & x_1\le 1000 &&\text{(silicio)},\\
 & x_2\le 1500 &&\text{(germanio)},\\
 & x_1+x_2\le 1750 &&\text{(plástico)},\\
 & 4x_1+2x_2\le 4800 &&\text{(cobre)},\\
 & x_1,x_2\ge 0 .
\end{aligned}
\]
En forma matricial, $\max\{c^\top x:Ax\le b,\ x\ge 0\}$.
La holgura $b_i-(Ax)_i$ mide cuánto queda del insumo $i$.
\end{frame}

\begin{frame}{Región factible y rectas de igual margen}
\begin{columns}[c,onlytextwidth]
\begin{column}{.47\textwidth}
Cada restricción conserva un semiplano. Su intersección es el polígono sombreado.
\medskip

La recta $12x_1+9x_2=z$ se desplaza hacia valores mayores de $z$.
El último contacto ocurre donde se agotan el plástico y el cobre.
\medskip

\[
\begin{aligned}
x^*&=(650,1100),\\
z^*&=17\,700 .
\end{aligned}
\]
\end{column}

\begin{frame}{Vértices óptimos y empates}
En un polígono factible acotado, cada plan es un promedio ponderado
de los vértices. El objetivo lineal toma el mismo promedio de sus valores.
\[
x=\sum_i\theta_i v_i,\qquad
c^\top x=\sum_i\theta_i\,c^\top v_i,\qquad
\theta_i\ge 0,\quad \sum_i\theta_i=1 .
\]
\concepto{Al menos un vértice es óptimo.}
\medskip

Si el objetivo de BIM fuera $\max(x_1+x_2)$, cualquier punto de la arista
entre $(250,1500)$ y $(650,1100)$ alcanzaría $1750$.
En ambos extremos y entre ellos se agota el plástico.
\end{frame}

\begin{frame}{Regiones no acotadas}
\begin{columns}[c,onlytextwidth]
\begin{column}{.50\textwidth}
La región satisface $x_1,x_2\ge0$, $-x_1+x_2\le2$ y $-x_1+2x_2\le6$.
Sus dos rayos permiten avanzar indefinidamente.
\medskip

\textbf{Minimizar $x_1-3x_2$.} Cada paso en $(2,1)$ reduce el costo en $1$.
\medskip

\textbf{Minimizar $4x_1-x_2$.} Ambos rayos aumentan el costo. El mínimo
vale $-2$ en $(0,2)$.
\end{column}

\begin{frame}{Forma estándar y base inicial}
Para el símplex usaremos una minimización con igualdades y variables no negativas.
\[
\min_{w\ge0}\ \tilde c^\top w
\qquad \text{sujeto a}\qquad
\tilde A w=b .
\]
Para BIM, agregamos holguras $s=b-Ax$ y cambiamos el signo del objetivo.
\[
 w=(x,s),\qquad \tilde A=[A\mid I],\qquad \tilde c=(-c,0).
\]
\begin{block}{Plan inicial}
Con $x=0$, las holguras valen $s=b$. Las columnas de $I$ forman
la base inicial y el plan corresponde al origen.
\end{block}
\end{frame}

\begin{frame}{Soluciones básicas factibles}
Supongamos que $\tilde A$ tiene $m$ filas linealmente independientes.
Elegimos $m$ columnas independientes para formar $B$.
\[
 w_{\mathcal N}=0,\qquad B w_{\mathcal B}=b.
\]
\begin{itemize}
\item Si $w_{\mathcal B}\ge0$, el punto es una \concepto{solución básica factible}.
\item Estas soluciones son los vértices de la región factible.
\item Si alguna variable básica vale cero, el punto es degenerado.
Un pivote puede cambiar la base en el mismo vértice.
\end{itemize}
\end{frame}

\begin{frame}{Costos reducidos y variable entrante}
Si una variable no básica $w_j$ sube de $0$ a $t$, las básicas deben
ajustarse para conservar $\tilde A w=b$.
\[
 B\bar b=b,\qquad Bu=a_j,\qquad
 w_{\mathcal B}(t)=\bar b-tu .
\]
El cambio en el objetivo es $t\bar c_j$, donde
\[
 B^\top y=\tilde c_{\mathcal B},\qquad
 \boxed{\bar c_j=\tilde c_j-a_j^\top y.}
\]
Si $\bar c_j<0$, cualquier paso positivo factible reduce el costo.
Si todos los costos reducidos son no negativos, la solución básica es óptima.
\end{frame}

\begin{frame}{Cociente mínimo y primer pivote de BIM}
La variable que sale es la primera básica que llega a cero.
\[
 t_*=\min_{i:u_i>0}\frac{\bar b_i}{u_i}.
\]
En BIM, con $B=I$, los costos reducidos de $x_1,x_2$ son $-12$ y $-9$.
La regla de Bland elige $x_1$, el menor índice con costo reducido negativo.
\medskip

\begin{center}
\begin{tabular}{lccc}
\toprule
Insumo que se consume & Silicio & Plástico & Cobre\\
\midrule
Cociente & $1000/1$ & $1750/1$ & $4800/4$\\
\bottomrule
\end{tabular}
\end{center}
\medskip

El silicio se agota primero. Entra $x_1$ y sale su holgura.
\end{frame}

\begin{frame}{Algoritmo del símplex revisado}
\begin{enumerate}
\item Resolver $B\bar b=b$ y $B^\top y=\tilde c_{\mathcal B}$.
\item Calcular los costos reducidos. Si todos son no negativos,
      devolver la solución básica.
\item Elegir un $\bar c_j<0$ y resolver $Bu=a_j$. Si $u\le0$,
      el objetivo puede disminuir indefinidamente.
\item Aplicar el cociente mínimo, cambiar la columna de la base y repetir.
\end{enumerate}
\medskip

\textbf{Regla de Bland.} Elegir el menor índice elegible para entrar
y el menor índice básico si hay empate al salir.
\medskip

\textbf{Fase I.} Si falta una base factible, se minimiza la suma de
variables artificiales. Un mínimo positivo certifica infactibilidad.
\end{frame}

\begin{frame}{Certificado dual y precios sombra}
Si $\lambda\ge0$ y $A^\top\lambda\ge c$, cada plan factible de BIM cumple
\[
 c^\top x\le\lambda^\top Ax\le b^\top\lambda .
\]
El dual minimiza $b^\top\lambda$ sobre estos valores unitarios.
\medskip

Para BIM, $\lambda=(0,0,6,1.5)$ da la cota
$6(1750)+1.5(4800)=17\,700$. El plan $(650,1100)$ alcanza esa cota.
\begin{block}{Una unidad adicional de plástico}
Con $b_3=1751$, el nuevo plan es $(649,1102)$ y el margen sube a
$17\,706$. El precio sombra de $6$ mide ese aumento local.
\end{block}
\fuente{\libroPL{06-bim-dual.html}{Postek et al. (2025), cuaderno 2.6}}
\end{frame}

\begin{frame}[fragile]{BIM con \texttt{linprog} y HiGHS}
\begin{lstlisting}
from scipy.optimize import linprog

A = [[1, 0], [0, 1], [1, 1], [4, 2]]
b = [1000, 1500, 1750, 4800]
res = linprog([-12, -9], A_ub=A, b_ub=b,
              bounds=(0, None), method="highs")
if not res.success:
    raise RuntimeError(res.message)

print(res.x, -res.fun)         # [650. 1100.] 17700.0
print(-res.ineqlin.marginals)   # [0. 0. 6. 1.5]
\end{lstlisting}
\medskip

SciPy minimiza. Se cambia el signo del objetivo al ingresar los datos
y el de los marginales al interpretar el máximo.
\fuente{\href{https://docs.scipy.org/doc/scipy/reference/generated/scipy.optimize.linprog.html}{\textcolor{upazul}{Documentación de linprog}}}
\end{frame}

\begin{frame}{Regresión LAD}
LAD (\emph{Least Absolute Deviation}) es la regresión de mínima suma de
desviaciones absolutas. Para el hogar $i$, $y_i$ es el gasto y $z_i$ es
el vector de regresores.
\[
\begin{aligned}
\min\quad &\sum_i(p_i+q_i)\\
\text{sujeto a}\quad
 &\alpha+z_i^\top\beta+p_i-q_i=y_i \quad (\text{cada } i),\\
 &p_i,q_i\ge0,\qquad \alpha,\beta_j\text{ libres}.
\end{aligned}
\]
En el óptimo, $p_i$ y $q_i$ son las partes positiva y negativa del residuo.
Si ambas fueran positivas, se podrían reducir por igual manteniendo el ajuste.
Con solo un intercepto, el ajuste LAD estima una mediana.
\fuente{\libroPL{02-lad-regression.html\#model}{Postek et al. (2025), cuaderno 2.2}}
\end{frame}

\begin{frame}{Cartera MAD}
MAD (\emph{Mean Absolute Deviation}) significa desviación absoluta media.
\smallskip

$w_j$ es el peso del activo $j$, $r_{tj}$ su retorno observado en $t$ y
$\bar r_j$ su retorno medio muestral.
\[
\begin{aligned}
\min\quad &\frac1T\sum_{t=1}^T(p_t+q_t)\\
\text{sujeto a}\quad
 &p_t-q_t=\sum_j(r_{tj}-\bar r_j)w_j \quad(\text{cada }t),\\
 &\sum_j\bar r_jw_j\ge R,\qquad \sum_j w_j=1,\\
 &0\le w_j\le u_j,\qquad p_t,q_t\ge0.
\end{aligned}
\]
Se minimiza el MAD de los retornos de la cartera respecto de su media.
$R$ fija el retorno medio mínimo y $u_j$ limita el peso del activo $j$.
\fuente{\libroPL{03-mad-portfolio-optimization.html\#mean-absolute-deviation-mad-portfolio-optimization}{Postek et al. (2025), cuaderno 2.3}}
\end{frame}
