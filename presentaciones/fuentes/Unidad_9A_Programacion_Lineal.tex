\begin{frame}{Variables, objetivo y restricciones}
$x_1,x_2$ son las cantidades de chips de cada tipo. El modelo continuo es
\[
\begin{aligned}
\max \quad &12x_1+9x_2\\
\text{sujeto a}\quad
 & x_1\le 1000 &&\text{(silicio)},\\
 & x_2\le 1500 &&\text{(germanio)},\\
 & x_1+x_2\le 1750 &&\text{(plástico)},\\
 & 4x_1+2x_2\le 4800 &&\text{(cobre)},\\
 & x_1,x_2\ge 0 .
\end{aligned}
\]
En forma matricial, $\max\{c_0^\top x:A_0x\le b,\ x\ge 0\}$.
La holgura $b_i-(A_0x)_i$ mide cuánto queda del insumo $i$.
\end{frame}

\medskip

\begin{frame}{Vértices óptimos y empates}
En un polígono factible acotado, cada plan es un promedio ponderado
de los vértices. El objetivo lineal toma el mismo promedio de sus valores.
\[
x=\sum_i\theta_i v_i,\qquad
c_0^\top x=\sum_i\theta_i\,c_0^\top v_i,\qquad
\theta_i\ge 0,\quad \sum_i\theta_i=1 .
\]
\concepto{Al menos un vértice es óptimo.}
\medskip

Si el objetivo de BIM fuera $\max(x_1+x_2)$, cualquier punto de la arista
entre $(250,1500)$ y $(650,1100)$ alcanzaría $1750$.
En ambos extremos y entre ellos se agota el plástico.
\end{frame}

\begin{frame}{Regiones no acotadas}
\begin{columns}[c,onlytextwidth]
\begin{column}{.50\textwidth}
La región satisface $x_1,x_2\ge0$, $-x_1+x_2\le2$ y $-x_1+2x_2\le6$.
Sus dos rayos permiten avanzar indefinidamente.
\medskip

\textbf{Minimizar $x_1-3x_2$.} Cada paso en $(2,1)$ reduce el costo en $1$.
\medskip

\textbf{Minimizar $4x_1-x_2$.} Ambos rayos aumentan el costo. El mínimo
vale $-2$ en $(0,2)$.
\end{column}

\begin{frame}{Forma estándar y base inicial}
En la notación de Bazaraa, la forma estándar es
\[
\min_{x\ge0}\ c^\top x \qquad \text{sujeto a}\qquad Ax=b,
\qquad A\in\mathbb R^{m\times n},\quad \operatorname{rang}(A)=m.
\]
Para BIM, $x_1,x_2$ son los niveles de producción y $x_3,\ldots,x_6$
son las holguras de silicio, germanio, plástico y cobre, en ese orden.
\[
 A=[A_0\mid I],\qquad c=\begin{pmatrix}-c_0\\0\end{pmatrix}.
\]
\begin{block}{Plan inicial}
Con $x_1=x_2=0$, las holguras valen $b$. La matriz básica inicial es
$B=[a_3\mid a_4\mid a_5\mid a_6]=I$, donde $a_j$ es la columna $j$ de $A$.
\end{block}
\end{frame}

\begin{frame}{Soluciones básicas factibles}
Reordenamos las columnas de $A$ para reunir en $B$ las $m$ columnas
elegidas, con $B$ invertible. Las demás forman $N$.
\[
 A=[B\mid N],\qquad Bx_B+Nx_N=b,\qquad
 x_N=0\ \Longrightarrow\ x_B=B^{-1}b .
\]
Las variables de $x_B$ son \concepto{básicas}. Las de $x_N$ son
\concepto{no básicas}. La solución básica es factible si $x_B\ge0$.
Las soluciones básicas factibles son los vértices. Si alguna básica vale
cero, el vértice es degenerado.
\begin{block}{Ejemplo 3.2}
Con $x_1+x_2+x_3=6$ y $x_2+x_4=3$, elegir $B=[a_1\mid a_2]$
fija $x_3=x_4=0$. Se obtiene $x=(3,3,0,0)$, que se proyecta en el
vértice $(3,3)$ del plano $(x_1,x_2)$.
\end{block}
\end{frame}

\begin{frame}{Costos reducidos y variable entrante}
Desde la solución básica factible $x_B=B^{-1}b$, $x_N=0$,
despejamos las básicas al permitir que cambie $x_N$.
\[
 Bx_B+Nx_N=b
 \quad\Longrightarrow\quad x_B=B^{-1}b-B^{-1}Nx_N .
\]
Al sustituir en el objetivo, con $R$ el conjunto de índices no básicos,
\[
\begin{aligned}
 z&=c_B^\top x_B+c_N^\top x_N
   =z_0+\sum_{j\in R}\bar c_jx_j,\\
 z_0&=c_B^\top B^{-1}b,\qquad
 z_j=c_B^\top B^{-1}a_j,\qquad \boxed{\bar c_j=c_j-z_j}.
\end{aligned}
\]
El costo reducido $\bar c_j$ mide el cambio por unidad de $x_j$.
Si $\bar c_j<0$, conviene aumentar $x_j$ siempre que el paso sea factible.
Si $\bar c_j\ge0$ para todo $j\in R$, la solución básica actual es óptima.
\end{frame}

\begin{frame}{Cociente mínimo y primer pivote de BIM}
Al aumentar solo la entrante $x_k$, la básica de la fila $i$ queda en
\[
 x_{B_i}=\bar b_i-y_{ik}x_k,\qquad \bar b=B^{-1}b,\qquad
 y_{ik}=(B^{-1}a_k)_i.
\]
Solo disminuyen las básicas con $y_{ik}>0$. Para conservar $x_B\ge0$,
\[
 0\le x_k\le \bar x_k=\min_{i:y_{ik}>0}\frac{\bar b_i}{y_{ik}} .
\]
En BIM, $\bar c_1=-12$ y $\bar c_2=-9$. Bland elige $k=1$.
Como $B=I$, $B^{-1}a_1=(1,0,1,4)^\top$ y
\[
 \bar x_1=\min\{1000/1,\ 1750/1,\ 4800/4\}=1000 .
\]
El mínimo está en la fila $r=1$. Sale $x_3$ y $a_1$ sustituye a $a_3$ en $B$.
\end{frame}

\begin{frame}{Algoritmo del símplex revisado}
\begin{enumerate}
\item Resolver $B\bar b=b$ y calcular $\bar c_j=c_j-z_j$ para $j\in R$.
\item Si todos los $\bar c_j\ge0$, devolver la solución básica actual.
\item Elegir $k$ con $\bar c_k<0$. Obtener $y_{ik}=(B^{-1}a_k)_i$.
      Si todos los $y_{ik}\le0$, el costo puede disminuir sin límite.
\item Calcular $\bar x_k$ y la fila saliente $r$. Sustituir $a_{B_r}$
      por $a_k$ en $B$ y repetir. Si $\bar x_k=0$, el punto no se mueve.
\end{enumerate}
\medskip

\textbf{Regla de Bland.} Elegir el menor índice elegible para entrar
y el menor índice básico si hay empate al salir.
\medskip

\textbf{Fase I.} Si falta una base factible, se minimiza la suma de
variables artificiales. Un mínimo positivo certifica infactibilidad.
\medskip

El código evalúa las expresiones con $B^{-1}$ mediante sistemas lineales.
\end{frame}

\begin{frame}{Certificado dual y precios sombra}
Volvemos al máximo original de BIM. Si $\lambda\ge0$ y $A_0^\top\lambda\ge c_0$,
cada plan factible satisface
\[
 c_0^\top x\le\lambda^\top A_0x\le b^\top\lambda .
\]
El dual minimiza $b^\top\lambda$ sobre estos valores unitarios.
\medskip

Para BIM, $\lambda=(0,0,6,1.5)$ da la cota
$6(1750)+1.5(4800)=17\,700$. El plan $(650,1100)$ alcanza esa cota.
\begin{block}{Una unidad adicional de plástico}
Con $b_3=1751$, el nuevo plan es $(649,1102)$ y el margen sube a
$17\,706$. El precio sombra de $6$ mide ese aumento local.
\end{block}
\fuente{\libroPL{06-bim-dual.html}{Postek et al. (2025), cuaderno 2.6}}
\end{frame}

\begin{frame}[fragile]{BIM con \texttt{linprog} y HiGHS}
\begin{lstlisting}
from scipy.optimize import linprog

A = [[1, 0], [0, 1], [1, 1], [4, 2]]
b = [1000, 1500, 1750, 4800]
res = linprog([-12, -9], A_ub=A, b_ub=b,
              bounds=(0, None), method="highs")
if not res.success:
    raise RuntimeError(res.message)

print(res.x, -res.fun)         # [650. 1100.] 17700.0
print(-res.ineqlin.marginals)   # [0. 0. 6. 1.5]
\end{lstlisting}
\medskip

SciPy minimiza. Se cambia el signo del objetivo al ingresar los datos
y el de los marginales al interpretar el máximo.
\fuente{\href{https://docs.scipy.org/doc/scipy/reference/generated/scipy.optimize.linprog.html}{\textcolor{upazul}{Documentación de linprog}}}
\end{frame}

\begin{frame}{Regresión LAD}
LAD (\emph{Least Absolute Deviation}) es la regresión de mínima suma de
desviaciones absolutas. Para el hogar $i$, $y_i$ es el gasto y $z_i$ es
el vector de regresores.
\[
\begin{aligned}
\min\quad &\sum_i(p_i+q_i)\\
\text{sujeto a}\quad
 &\alpha+z_i^\top\beta+p_i-q_i=y_i \quad (\text{cada } i),\\
 &p_i,q_i\ge0,\qquad \alpha,\beta_j\text{ libres}.
\end{aligned}
\]
En el óptimo, $p_i$ y $q_i$ son las partes positiva y negativa del residuo.
Si ambas fueran positivas, se podrían reducir por igual manteniendo el ajuste.
Con solo un intercepto, el ajuste LAD estima una mediana.
\fuente{\libroPL{02-lad-regression.html\#model}{Postek et al. (2025), cuaderno 2.2}}
\end{frame}

\begin{frame}{Cartera MAD}
MAD (\emph{Mean Absolute Deviation}) significa desviación absoluta media.
\smallskip

$w_j$ es el peso del activo $j$, $r_{tj}$ su retorno observado en $t$ y
$\bar r_j$ su retorno medio muestral.
\[
\begin{aligned}
\min\quad &\frac1T\sum_{t=1}^T(p_t+q_t)\\
\text{sujeto a}\quad
 &p_t-q_t=\sum_j(r_{tj}-\bar r_j)w_j \quad(\text{cada }t),\\
 &\sum_j\bar r_jw_j\ge R,\qquad \sum_j w_j=1,\\
 &0\le w_j\le u_j,\qquad p_t,q_t\ge0.
\end{aligned}
\]
Se minimiza el MAD de los retornos de la cartera respecto de su media.
$R$ fija el retorno medio mínimo y $u_j$ limita el peso del activo $j$.
\fuente{\libroPL{03-mad-portfolio-optimization.html\#mean-absolute-deviation-mad-portfolio-optimization}{Postek et al. (2025), cuaderno 2.3}}
\end{frame}
