\begin{frame}{Variables, objetivo y restricciones}
$x_1,x_2$ son las cantidades de chips de cada tipo. El modelo continuo es
\[
\begin{aligned}
\max \quad &12x_1+9x_2\\
\text{sujeto a}\quad
 & x_1\le 1000 &&\text{(silicio)},\\
 & x_2\le 1500 &&\text{(germanio)},\\
 & x_1+x_2\le 1750 &&\text{(plástico)},\\
 & 4x_1+2x_2\le 4800 &&\text{(cobre)},\\
 & x_1,x_2\ge 0 .
\end{aligned}
\]
En forma matricial, $\max\{c_0^\top x:A_0x\le b,\ x\ge 0\}$.
La holgura $b_i-(A_0x)_i$ mide cuánto queda del insumo $i$.
\end{frame}

\begin{frame}{Vértices óptimos y empates}
En un polígono factible acotado, cada plan es un promedio ponderado
de los vértices. El objetivo lineal toma el mismo promedio de sus valores.
\[
x=\sum_i\theta_i v_i,\qquad
c_0^\top x=\sum_i\theta_i\,c_0^\top v_i,\qquad
\theta_i\ge 0,\quad \sum_i\theta_i=1 .
\]
\concepto{Al menos un vértice es óptimo.}
\medskip

Si el objetivo de BIM fuera $\max(x_1+x_2)$, cualquier punto de la arista
entre $(250,1500)$ y $(650,1100)$ alcanzaría $1750$.
En ambos extremos y entre ellos se agota el plástico.
\end{frame}

\begin{frame}{Regiones no acotadas}
\begin{columns}[c,onlytextwidth]
\begin{column}{.50\textwidth}

Sus dos \hyperlink{teoria-rayos}{\concepto{rayos de recesión}}
permiten avanzar indefinidamente.
\medskip

\textbf{Minimizar $x_1-3x_2$.} Cada paso en $(2,1)$ reduce el costo en $1$.
\medskip

\textbf{Minimizar $4x_1-x_2$.} Ambos rayos aumentan el costo. El mínimo
vale $-2$ en $(0,2)$.
\end{column}

\begin{frame}{Forma estándar y base inicial}
En la notación de Bazaraa, la forma estándar es
\[
\min_{x\ge0}\ c^\top x \qquad \text{sujeto a}\qquad Ax=b,
\qquad A\in\mathbb R^{m\times n},\quad \operatorname{rang}(A)=m.
\]
Para BIM, $x_1,x_2$ son los niveles de producción y $x_3,\ldots,x_6$
son las holguras de silicio, germanio, plástico y cobre, en ese orden.
\[
 A=[A_0\mid I],\qquad c=\begin{pmatrix}-c_0\\0\end{pmatrix}.
\]
\begin{block}{Plan inicial}
Con $x_1=x_2=0$, las holguras valen $b$. La matriz básica inicial es
$B=[a_3\mid a_4\mid a_5\mid a_6]=I$, donde $a_j$ es la columna $j$ de $A$.
\end{block}
\end{frame}

\begin{frame}{Soluciones básicas factibles}
Reordenamos las columnas de $A$ para reunir en $B$ las $m$ columnas
elegidas, con $B$ invertible. Las demás forman $N$.
\[
 A=[B\mid N],\qquad Bx_B+Nx_N=b,\qquad
 x_N=0\ \Longrightarrow\ x_B=B^{-1}b .
\]
Las variables de $x_B$ son \concepto{básicas}. Las de $x_N$ son
\concepto{no básicas}. La solución básica es factible si $x_B\ge0$.
Las soluciones básicas factibles son los vértices
(\hyperlink{teoria-sbf}{\concepto{demostración}}).
Si alguna básica vale cero, la solución es degenerada.
\begin{block}{Ejemplo 3.2}
Con $x_1+x_2+x_3=6$ y $x_2+x_4=3$, elegir $B=[a_1\mid a_2]$
fija $x_3=x_4=0$. Se obtiene $x=(3,3,0,0)$, que se proyecta en el
vértice $(3,3)$ del plano $(x_1,x_2)$.
\end{block}
\end{frame}

\begin{frame}{Costos reducidos y variable entrante}
Desde la solución básica factible $x_B=B^{-1}b$, $x_N=0$,
despejamos las básicas al permitir que cambie $x_N$.
\[
 Bx_B+Nx_N=b
 \quad\Longrightarrow\quad x_B=B^{-1}b-B^{-1}Nx_N .
\]
Al sustituir en el objetivo, con $R$ el conjunto de índices no básicos,
\[
\begin{aligned}
 z&=c_B^\top x_B+c_N^\top x_N
   =z_0+\sum_{j\in R}\bar c_jx_j,\\
 z_0&=c_B^\top B^{-1}b,\qquad
 z_j=c_B^\top B^{-1}a_j,\qquad \boxed{\bar c_j=c_j-z_j}.
\end{aligned}
\]
El costo reducido $\bar c_j$ mide el cambio por unidad de $x_j$.
Si $\bar c_j<0$, conviene aumentar $x_j$ siempre que el paso sea factible.
Si $\bar c_j\ge0$ para todo $j\in R$, la solución básica actual es óptima
(\hyperlink{teoria-certificados}{\concepto{demostración}}).
\end{frame}

\begin{frame}{Cociente mínimo y primer pivote de BIM}
Al aumentar solo la entrante $x_k$, la básica de la fila $i$ queda en
\[
 x_{B_i}=\bar b_i-y_{ik}x_k,\qquad \bar b=B^{-1}b,\qquad
 y_{ik}=(B^{-1}a_k)_i.
\]
Solo disminuyen las básicas con $y_{ik}>0$. Para conservar $x_B\ge0$,
\[
 0\le x_k\le \bar x_k=\min_{i:y_{ik}>0}\frac{\bar b_i}{y_{ik}} .
\]
En BIM, $\bar c_1=-12$ y $\bar c_2=-9$. Bland elige $k=1$.
Como $B=I$, $B^{-1}a_1=(1,0,1,4)^\top$ y
\[
 \bar x_1=\min\{1000/1,\ 1750/1,\ 4800/4\}=1000 .
\]
El mínimo está en la fila $r=1$. Sale $x_3$ y $a_1$ sustituye a $a_3$ en $B$
(\hyperlink{teoria-pivote}{\concepto{justificación}}).
\end{frame}

\begin{frame}{Algoritmo del símplex revisado}
\begin{enumerate}
\item Resolver $B\bar b=b$ y calcular $\bar c_j=c_j-z_j$ para $j\in R$.
\item Si todos los $\bar c_j\ge0$, devolver la solución básica actual.
\item Elegir $k$ con $\bar c_k<0$. Obtener $y_{ik}=(B^{-1}a_k)_i$.
      Si todos los $y_{ik}\le0$, el costo puede disminuir sin límite.
\item Calcular $\bar x_k$ y la fila saliente $r$. Sustituir $a_{B_r}$
      por $a_k$ en $B$ y repetir. Si $\bar x_k=0$, el punto no se mueve.
\end{enumerate}
\medskip

\hyperlink{teoria-terminacion}{\concepto{Regla de Bland.}} Elegir el menor índice elegible para entrar
y el menor índice básico si hay empate al salir.
\medskip

\hyperlink{teoria-fase1}{\concepto{Fase I.}} Si falta una base factible, se minimiza la suma de
variables artificiales. Un mínimo positivo certifica infactibilidad.
\medskip

El código evalúa las expresiones con $B^{-1}$ mediante sistemas lineales.
\end{frame}

\begin{frame}{Certificado dual y precios sombra}
Volvemos al máximo original de BIM. Si $\lambda\ge0$ y $A_0^\top\lambda\ge c_0$,
cada plan factible satisface
\[
 c_0^\top x\le\lambda^\top A_0x\le b^\top\lambda .
\]
El dual minimiza $b^\top\lambda$ sobre estos valores unitarios.
\medskip

Para BIM, $\lambda=(0,0,6,1.5)$ da la cota
$6(1750)+1.5(4800)=17\,700$. El plan $(650,1100)$ alcanza esa cota.
\begin{block}{Una unidad adicional de plástico}
Con $b_3=1751$, el nuevo plan es $(649,1102)$ y el margen sube a
$17\,706$. El precio sombra de $6$ mide ese aumento local.
\end{block}
\fuente{\libroPL{06-bim-dual.html}{Postek et al. (2025), cuaderno 2.6}}
\end{frame}

\begin{frame}{Regresión LAD}
LAD (\emph{Least Absolute Deviation}) es la regresión de mínima suma de
desviaciones absolutas. Para el hogar $i$, $y_i$ es el gasto y $z_i$ es
el vector de regresores.
\[
\begin{aligned}
\min\quad &\sum_i(p_i+q_i)\\
\text{sujeto a}\quad
 &\alpha+z_i^\top\beta+p_i-q_i=y_i \quad (\text{cada } i),\\
 &p_i,q_i\ge0,\qquad \alpha,\beta_j\text{ libres}.
\end{aligned}
\]
En el óptimo, $p_i$ y $q_i$ son las partes positiva y negativa del residuo.
Si ambas fueran positivas, se podrían reducir por igual manteniendo el ajuste.
Con solo un intercepto, el ajuste LAD estima una mediana.
\fuente{\libroPL{02-lad-regression.html\#model}{Postek et al. (2025), cuaderno 2.2}}
\end{frame}

\begin{frame}{Cartera MAD}
MAD (\emph{Mean Absolute Deviation}) significa desviación absoluta media.
\smallskip

$w_j$ es el peso del activo $j$, $r_{tj}$ su retorno observado en $t$ y
$\bar r_j$ su retorno medio muestral.
\[
\begin{aligned}
\min\quad &\frac1T\sum_{t=1}^T(p_t+q_t)\\
\text{sujeto a}\quad
 &p_t-q_t=\sum_j(r_{tj}-\bar r_j)w_j \quad(\text{cada }t),\\
 &\sum_j\bar r_jw_j\ge R,\qquad \sum_j w_j=1,\\
 &0\le w_j\le u_j,\qquad p_t,q_t\ge0.
\end{aligned}
\]
Se minimiza el MAD de los retornos de la cartera respecto de su media.
$R$ fija el retorno medio mínimo y $u_j$ limita el peso del activo $j$.
\fuente{\libroPL{03-mad-portfolio-optimization.html\#mean-absolute-deviation-mad-portfolio-optimization}{Postek et al. (2025), cuaderno 2.3}}
\end{frame}

\section{Fundamentos del símplex}

\begin{frame}[label=teoria-convexidad]{Convexidad y puntos extremos}
En forma estándar, $P=\{x\in\mathbb R^n:Ax=b,\ x\ge0\}$ es convexo.
Si $u,v\in P$ y $0\le\lambda\le1$, entonces
\[
 A\bigl(\lambda u+(1-\lambda)v\bigr)=b,\qquad
 \lambda u+(1-\lambda)v\ge0.
\]
Un \concepto{punto extremo} de $P$ no puede escribirse como
$\lambda u+(1-\lambda)v$ con $u,v\in P$, $u\ne v$ y $0<\lambda<1$.
\medskip

La \concepto{envolvente convexa} $\operatorname{conv}(S)$ reúne las
combinaciones finitas $\sum_i\lambda_i x^i$ de puntos de $S$ con
$\lambda_i\ge0$ y $\sum_i\lambda_i=1$.
\begin{block}{Teorema de Carathéodory}
En $\mathbb R^n$, basta combinar a lo sumo $n+1$ puntos de $S$ para
representar un punto de $\operatorname{conv}(S)$.
\end{block}
\end{frame}

\begin{frame}[label=teoria-rayos]{Dirección factible y recesión}
Fijemos $x\in P$. Una \concepto{dirección factible} $d$ permite
$x+\theta d\in P$ para todo $\theta\in[0,\varepsilon]$ con algún
$\varepsilon>0$. En forma estándar esto equivale a
\[
 Ad=0,\qquad d_i\ge0 \text{ cuando }x_i=0.
\]
En efecto, $Ad=0$ conserva la igualdad y una componente positiva
puede disminuir durante un intervalo corto. Si $d_i<0$, la cota
$\theta\le x_i/(-d_i)$ evita que esa componente se vuelva negativa.
\medskip

Una \concepto{dirección de recesión} admite todo $\theta\ge0$.
Cuando $P\ne\varnothing$, su cono de recesión es
\[
 \operatorname{rec}(P)=\{d:Ad=0,\ d\ge0\}.
\]
Si $c^\top d<0$ en ese cono, el costo $c^\top x+\theta c^\top d$
tiende a $-\infty$.
\end{frame}

\begin{frame}{Descomposición del poliedro y óptimo finito}
Para $P$ no vacío en forma estándar, sean $x^1,\ldots,x^p$ sus vértices
y $d^1,\ldots,d^q$ sus direcciones extremas de recesión. Todo
punto factible tiene una representación de la forma
\[
 x=\sum_{i=1}^p\lambda_i x^i+\sum_{j=1}^q\mu_j d^j,\qquad
 \lambda_i,\mu_j\ge0,\qquad \sum_{i=1}^p\lambda_i=1 .
\]
Al aplicar $c^\top$, la primera suma es un promedio de costos de vértices.
\begin{itemize}
\item Si algún $c^\top d^j<0$, se puede aumentar $\mu_j$ sin límite y
      el problema es no acotado inferiormente.
\item Si todos los $c^\top d^j\ge0$, la segunda suma aporta un costo
      no negativo. Cualquier vértice de menor costo alcanza el mínimo.
\end{itemize}
\medskip

Así, con un óptimo finito, \concepto{existe al menos un vértice óptimo}.
Una arista de costo constante puede contener otros óptimos.
\end{frame}

\begin{frame}[label=teoria-sbf]{De solución básica factible a vértice}
Sea $x$ una solución básica factible con $x_B=B^{-1}b$ y $x_N=0$.
Supongamos que
\[
 x=\lambda u+(1-\lambda)v,\qquad
 u,v\in P,\qquad 0<\lambda<1.
\]
Las componentes no básicas satisfacen
\[
 0=\lambda u_N+(1-\lambda)v_N.
\]
Como $u_N,v_N\ge0$ y ambos pesos son positivos, se tiene $u_N=v_N=0$.
La factibilidad de $u$ y $v$ exige entonces
\[
 Bu_B=b=Bv_B.
\]
La matriz $B$ es invertible, de modo que $u_B=v_B=B^{-1}b$.
Por tanto, $u=v=x$. Esto prueba que $x$ es un punto extremo.
\end{frame}

\begin{frame}{De vértice a solución básica factible}
Sea $x$ un vértice y sea $J=\{j:x_j>0\}$. Si las columnas $a_j$
con $j\in J$ fueran dependientes, habría $\gamma\ne0$ tal que
\[
 A\gamma=0,\qquad \gamma_j=0 \text{ para } j\notin J.
\]
Para $\varepsilon>0$ suficientemente pequeño,
$x+\varepsilon\gamma$ y $x-\varepsilon\gamma$ serían factibles.
Son puntos distintos con promedio $x$, lo cual contradice que
$x$ sea extremo.
\medskip

Las columnas asociadas a las componentes positivas son, por tanto,
independientes. Como $\operatorname{rang}(A)=m$, se pueden completar
con otras columnas de $A$ hasta formar una base $B$. Las variables
añadidas valen cero. Todas las no básicas también valen cero, así que
\[
 x_N=0,\qquad Bx_B=b,\qquad x_B\ge0 .
\]
\end{frame}

\begin{frame}{Existencia de una solución básica factible}
Partimos de cualquier $x\in P$. Si las columnas correspondientes a
$x_j>0$ son dependientes, tomamos $\gamma\ne0$ con $A\gamma=0$ y
$\gamma_j=0$ donde $x_j=0$. Elegimos su signo para que alguna
$\gamma_j>0$.
\[
 \lambda=\min_{j:\gamma_j>0}\frac{x_j}{\gamma_j},\qquad
 x'=x-\lambda\gamma .
\]
\begin{itemize}
\item $Ax'=b$ porque $A\gamma=0$.
\item $x'\ge0$ por la elección del cociente mínimo.
\item Al menos una componente positiva se hace cero. Las que ya
      valían cero siguen en cero.
\end{itemize}
\medskip

Al repetir, disminuye el número de componentes positivas. Cuando sus
columnas son independientes, se completan hasta una base de $m$ columnas
como en la demostración anterior. Así, $P\ne\varnothing$ implica que existe
una solución básica factible.
\end{frame}

\begin{frame}[label=teoria-certificados]{Certificados de optimalidad y no acotación}
Para cualquier punto factible, el mismo despeje de la base da
\[
 z-z_0=\sum_{j\in R}\bar c_j x_j.
\]
Si todos los $\bar c_j\ge0$, la suma es no negativa porque $x_j\ge0$.
La solución básica actual alcanza $z_0$ y es óptima en toda la región.
\medskip

Si $\bar c_k<0$ y todos los $y_{ik}\le0$, definimos una dirección $d$
con $d_k=1$, $d_j=0$ para $j\in R\setminus\{k\}$ y
$d_B=-B^{-1}a_k$. Entonces
\[
 Ad=Bd_B+a_k=0,\qquad d\ge0,\qquad
 c^\top d=c_k-c_B^\top B^{-1}a_k=\bar c_k<0.
\]
Es una dirección de recesión que reduce el costo sin límite.
Si algún $y_{ik}>0$, el cociente acota este desplazamiento y puede dar
un paso de longitud cero.
\end{frame}

\begin{frame}[label=teoria-pivote]{Factibilidad y cambio de base en el pivote}
Con una entrante $k$ tal que $\bar c_k<0$, la prueba del cociente
da $\bar x_k=\bar b_r/y_{rk}$ y $y_{rk}>0$. Al tomar $x_k=\bar x_k$,
se conserva $x\ge0$ y la básica $x_{B_r}$ se hace cero.
\medskip

Sea $B'$ la matriz obtenida al reemplazar la columna $r$ de $B$
por $a_k$. La matriz $B^{-1}B'$ es la identidad con su columna $r$
reemplazada por $B^{-1}a_k$. Por tanto,
\[
 \det(B^{-1}B')=y_{rk}\ne0
 \quad\Longrightarrow\quad B' \text{ es invertible}.
\]
Las nuevas no básicas valen cero, incluida $x_{B_r}$, que acaba de salir.
El punto es, por tanto, una solución básica factible para $B'$.
Como solo cambió una de las variables que eran no básicas,
\[
 z_{\mathrm{nuevo}}=z_0+\bar c_k\bar x_k .
\]
Hay descenso estricto cuando $\bar x_k>0$. Con $\bar x_k=0$,
cambia la base en el mismo punto.
\end{frame}

\begin{frame}{Degeneración y regla de Bland}
Consideremos $x_1+x_2+x_3=3$ y $x_2+x_4=3$, con $x\ge0$.
El mismo vértice $x=(0,3,0,0)$ admite las bases
\[
 B=[a_1\mid a_2]\qquad\text{y}\qquad B'=[a_3\mid a_2].
\]
En la primera, $x_1$ es básica y vale cero. En la segunda,
$x_3$ es básica y vale cero. En ambas, el vector $x$ es el mismo.
\medskip

Si en una fila $\bar b_i=0$ y $y_{ik}>0$, el cociente es cero.
El pivote cambia de base sin reducir $z$. Una sucesión de estos cambios
puede regresar a una base anterior.
\begin{block}{Regla de Bland}
Entre las no básicas con $\bar c_j<0$, entra la de menor índice.
En un empate del cociente, sale la básica de menor índice original.
En aritmética exacta, esta regla evita el ciclaje.
\end{block}
\end{frame}

\begin{frame}[label=teoria-terminacion]{Por qué termina el símplex}
Cada pivote produce otra base factible y hay a lo sumo
$\binom{n}{m}$ conjuntos de columnas que pueden formar una base.
\begin{itemize}
\item Con pasos positivos, el costo baja estrictamente. Volver a
      la misma base devolvería el mismo punto y el mismo costo.
\item En un ciclo, el costo tendría que quedar constante. Como
      $\bar c_k<0$ en cada iteración, todos los pasos del ciclo
      tendrían que ser cero. El teorema de Bland excluye ese ciclo.
\end{itemize}
Así, en aritmética exacta y con una base inicial factible, el algoritmo
termina con una solución óptima o una dirección de costo no acotado.
\begin{block}{Una base degenerada en un punto óptimo}
En $\min\{-x_2:x_1+x_2=0,\ x\ge0\}$, con $B=[a_1]$ se tiene
$\bar c_2=-1$, pero $\bar x_2=0$. El único punto es $(0,0)$.
Tras el pivote, $B'=[a_2]$ certifica el mismo punto con $\bar c_1=1$.
\end{block}
\end{frame}

\begin{frame}[label=teoria-fase1]{Fase I y fase II}
Si no conocemos una base factible, primero ajustamos los signos de las
filas para que $b\ge0$. Agregamos un vector de variables artificiales $a$
y resolvemos
\[
 \min\ \mathbf 1^\top a
 \qquad\text{sujeto a}\qquad Ax+a=b,\quad x,a\ge0.
\]
Las columnas de las artificiales forman la base inicial, con $x=0$ y
$a=b$.
\begin{itemize}
\item Si el mínimo de fase I es positivo, $Ax=b,\ x\ge0$ es infactible.
      Un punto factible del sistema original permitiría tomar $a=0$.
\item Si el mínimo es cero, todas las artificiales valen cero.
      Las artificiales básicas que queden se retiran con pivotes de
      paso cero o eliminando filas redundantes.
\end{itemize}
\medskip

Con una base del sistema original, la \concepto{fase II} restaura
el objetivo $c^\top x$ y continúa con el símplex.
\end{frame}

\begin{frame}{Fase I en un sistema pequeño}
Consideremos $x_1-x_2=1$, con $x\ge0$. Al agregar una artificial,
la fase I comienza en $(x_1,x_2,a_1)=(0,0,1)$.
\[
 \min a_1,\qquad x_1-x_2+a_1=1,\qquad x_1,x_2,a_1\ge0 .
\]
Manteniendo $x_2=0$, aumentar $x_1$ de $0$ a $1$ reduce $a_1$ de $1$
a $0$. Entra $x_1$ y sale $a_1$. La fase II parte de $(x_1,x_2)=(1,0)$.
\medskip

En cambio, para $-x_1=1$ con $x_1\ge0$, la fila auxiliar es
$-x_1+a_1=1$. Esto obliga a $a_1=1+x_1\ge1$, de modo que el
mínimo de fase I es $1$ y el sistema original es infactible.
\end{frame}
